Case study · Senpai Refiner

Interface simples. O controle fica no motor.

O Senpai foi reescrito para usar o MHL como seu motor harness: a experiência permanece familiar, enquanto cada etapa passa a ter governança, limites claros e uma escolha de IA mais eficiente.

senpai-workflow.mh
macroestrutura MHL
01agent SenpaiAgent {
02  engine: "cli/codex"
03  command: "codex"
04  args: ["exec", "--full-auto", "${prompt}"]
05}
06
07workflow SenpaiBrief max 12 {
08  input work_item: string
09  var artifacts = ["brief", "decisions", "diagram"]
10  var approved = false
11
12  step Discover timeout 2m {
13    context = SenpaiAgent.run(prompt: "map ${work_item}")
14  }
15
16  step Fanout timeout 4m {
17    spawn reviews = SenpaiAgent.run(prompt: "review ${item}")
18      for item in artifacts
19    wait reviews timeout: 3m
20  }
21
22  step Approval timeout 1m {
23    if (!approved) pause("human approval required")
24  }
25}
